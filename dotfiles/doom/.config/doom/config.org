* auto-resize Emacs font size on =ax-bee= with 4k/1080 monitors
#+begin_src elisp
(defvar ax/1080p-font-size 12
  "Font size for 1080p displays.")

(defvar ax/4k-font-size 18
  "Font size for 4K / high-resolution displays.")

(defvar ax/monitor-font--timer nil)
(defvar ax/monitor-font--last-size nil)

(defun ax/set-font-for-monitor (&optional frame)
  "Set doom font size based on the current monitor (poll-friendly)."
  (when (display-graphic-p)
    (let* ((frame (or frame (selected-frame)))
           (geometry (frame-monitor-attribute 'geometry frame))
           (scale (or (frame-monitor-attribute 'scale-factor frame) 1))
           (width (and geometry (nth 2 geometry)))
           (size (when width
                   (let ((effective-width (* width scale)))
                     (if (> effective-width 3000)
                         ax/4k-font-size
                       ax/1080p-font-size)))))

      (when (and size (not (equal size ax/monitor-font--last-size)))
        (setq ax/monitor-font--last-size size)
        (setq doom-font (font-spec :family "Hack Nerd Font" :size size))
        (when (fboundp 'doom/reload-font)
          (doom/reload-font))
        ;; Show in the echo area (bottom/statusline area)
        (message "Font changed: %s" size)))))

(let ((hn (string-trim (system-name))))
  (when (and hn (string-match-p "\\`ax-bee\\'" hn))
    ;; Apply once right now
    (when (display-graphic-p)
      (ax/set-font-for-monitor (selected-frame)))

    ;; Poll to detect monitor changes even when focus doesn't change (Hyprland)
    (when ax/monitor-font--timer
      (cancel-timer ax/monitor-font--timer))
    (setq ax/monitor-font--timer
          (run-with-timer 0 1.0 (lambda () (ax/set-font-for-monitor (selected-frame)))))))
#+end_src

** when to save
If you set the variable ‘bookmark-save-flag’ to 1, each command that
sets a bookmark will also save your bookmarks; this way, you don't lose
any bookmark values even if Emacs crashes.  The value, if a number, says
how many bookmark modifications should go by between saving.  If you set
this variable to ‘nil’, Emacs only saves bookmarks if you explicitly use
‘M-x bookmark-save’.
#+begin_src emacs-lisp

** Calendar setup to interact with radicale
*TL;DR* — =SPC j c c= opens a read-only month view of what is on the server.
=SPC j c s= syncs org and the calendar both ways. Events can be created
anywhere (emacs, thunderbird, phone) and all sync to radicale. Only sync from a
machine whose org files are current, or it can push stale text back and undo
edits made elsewhere.

Calendar on the radicale server, same account as the contacts. =calfw= only views
it, =org-caldav= does the two-way editing. Both read the password from
=~/.authinfo=.

*** viewing the calendar (read only): =SPC j c c=
#+begin_src elisp
(defun ax/open-calendar ()
  "Open a read-only view of the calendar on radicale."
  (interactive)
  (require 'calfw)
  (require 'calfw-ical)
  (calfw-ical-data-cache-clear-all)
  (calfw-open-calendar-buffer
   :contents-sources
   (list (calfw-ical-create-source
          "http://192.168.178.8:5232/ax/calendar/"
          "calendar"
          "IndianRed"))))
#+end_src

*** syncing org and calendar: =SPC j c s=
Syncs both ways. It stays manual because it writes in both directions.
=org-caldav-url= is the base address, =org-caldav-calendar-id= the collection href.
New server events land in the inbox, and the listed files are read and exported.
=org-caldav-save-directory= holds the sync state and sits in ~/org so
syncthing shares it between machines.
With two machines the sync state has to match the org files it describes. If
emacs has =todo.org= open with stale text (doom only reverts on focus),
the sync pushes the old version back and undoes edits made elsewhere.
Run =SPC b r= first.
#+begin_src elisp
(use-package! org-caldav
  :defer t
  :config
  (setq org-caldav-url "http://192.168.178.8:5232/ax"
        org-caldav-calendar-id "calendar"
        org-caldav-inbox "~/org/caldav-inbox.org"
        org-caldav-files '("~/org/todo.org")
        org-caldav-sync-direction 'twoway
        org-caldav-save-directory "~/org/.org-caldav/"
        org-caldav-backup-file "~/org/.org-caldav/backup.org"))
#+end_src

Without this the sync result buffer takes over the whole frame.
#+begin_src elisp
(set-popup-rule! "^\\*org caldav sync result" :size 0.3 :quit t :select nil)
#+end_src

org-caldav exports via org's icalendar exporter, so these settings decide what syncs.
The default =(todo-start)= alone creates no event at all. A scheduled time
or a deadline on a TODO only becomes an event when the list holds
=event-if-todo-not-done=, and without it a sync reports success while pushing
nothing.
The timezone has to be set by hand. Its default reads the =TZ= environment
variable, which is unset on these machines, and org-caldav then writes the
times with no timezone attached and leaves every client to guess.
If task sync is ever turned on, =todo-start= and =todo-due= have to go back
into these two lists, or a task arrives without a start and without a due date.
#+begin_src elisp
(setq org-icalendar-timezone "Europe/Berlin")
(setq org-icalendar-use-scheduled
      '(event-if-not-todo event-if-todo-not-done))
(setq org-icalendar-use-deadline
      '(event-if-not-todo event-if-todo-not-done))
#+end_src

(defun ax-tmr-notify-send (timer)
  "Announce finished TIMER via notify-send.
Fall back to `tmr-notification-notify' if notify-send is unavailable."
  (if (executable-find "notify-send")
      (call-process "notify-send" nil 0 nil
                    "-a" "Emacs TMR"
                    "-u" (symbol-name tmr-notification-urgency)
                    "TMR May Ring"
                    (or (tmr--timer-description timer) "Time is up!"))
    (tmr-notification-notify timer)))

(use-package! tmr

  :config

        tmr-notification-urgency 'normal)

*** Matrix: elfeed-goodies header colors
Upstream =the-matrix-theme.el= leaves =mode-line= background unspecified
and doesn't define any =powerline-*= faces. =elfeed-goodies= draws its
search + entry header bars with powerline, so those faces fall back to
powerline's default greys — three ugly grey blocks in the header bar.
Override them with matrix-palette darks; the 3-step ramp (=color-bg= →
=color-bg-alt= → =color-darkest=) keeps powerline's separator arrows
visible. Done as overrides here rather than in the theme file because we
keep =the-matrix-theme.el= byte-for-byte upstream.

(defun thanos/wtype-text (text)
  "Process TEXT for wtype, handling newlines properly."
  (let* ((has-final-newline (string-match-p "\n$" text))
         (lines (split-string text "\n"))
         (last-idx (1- (length lines))))
    (string-join
     (cl-loop for line in lines
              for i from 0
              collect (cond
                       ;; Last line without final newline
                       ((and (= i last-idx) (not has-final-newline))
                        (format "wtype -s 350 \"%s\"" 
                                (replace-regexp-in-string "\"" "\\\\\"" line)))
                       ;; Any other line
                       (t
                        (format "wtype -s 350 \"%s\" && wtype -k Return" 
                                (replace-regexp-in-string "\"" "\\\\\"" line)))))
     " && ")))

** org-publish
The nav bar lives in =nav.html= in the source directory on the VPS — edit it there, no
config change or tangle needed.

Publish with =SPC j v= (=ax/publish-site=), which forces a rebuild and drops the cached nav.
Nothing else picks up a nav edit: org republishes a page based on its own mtime and
never looks at =nav.html=, so a plain =M-x org-publish= silently keeps the old one.

Keep the links root-relative (=/index.html=) — pages under =freebsd/= get this same nav.
#+begin_src elisp
(defvar ax/vps0-src-dir "/ssh:vps:/usr/local/www/mysite-src/")

(defvar ax/vps0-nav-cache nil
  "Cached contents of nav.html, reset at the start of each publish run.")

(defun ax/vps0-nav-preamble (_info)
  "Return the site nav HTML, read from nav.html in `ax/vps0-src-dir'."
  (or ax/vps0-nav-cache
      (setq ax/vps0-nav-cache
            (let ((f (expand-file-name "nav.html" ax/vps0-src-dir)))
              (unless (file-readable-p f)
                (user-error "ax: nav.html not readable at %s" f))
              (with-temp-buffer
                (insert-file-contents f)
                (buffer-string))))))

(setq org-publish-project-alist
      `(("ax-vps0"
         :base-directory ,ax/vps0-src-dir
         :base-extension "org"
         :publishing-directory "/ssh:vps:/usr/local/www/mysite/"
         :publishing-function org-html-publish-to-html
         :recursive t
         :section-numbers nil
         :html-preamble ax/vps0-nav-preamble)
        ("ax-images"
         :base-directory ,(concat ax/vps0-src-dir "assets/")
         :base-extension "png\\|jpg\\|jpeg\\|gif\\|svg\\|webp"
         :publishing-directory "/ssh:vps:/usr/local/www/mysite/assets/"
         :recursive t
         :publishing-function org-publish-attachment)
        ("ax-website"
         :components ("ax-vps0" "ax-images"))))

(defun ax/publish-site ()
  "Publish the whole ax-website project (HTML + images), forcing a full rebuild.
Also drops the cached nav so nav.html edits are picked up."
  (interactive)
  (setq ax/vps0-nav-cache nil)
  (org-publish "ax-website" t))
#+end_src
